## Rollback

If any QA gate fails or downstream consumers reject v10:

```bash
# 1. Stage v10 row backup (latest TSV)
LATEST=$(ls -t /tmp/pgbackup/all_chapters_pre_v10_*.tsv | head -1)

psql "$DATABASE_URL" <<SQL
BEGIN;
CREATE TEMP TABLE _chap_v10_rollback (
  slug text PRIMARY KEY,
  kind text,
  order_key int,
  title text,
  body_md text,
  illustration_url text
);
\\COPY _chap_v10_rollback FROM '${LATEST}' WITH (FORMAT csv, DELIMITER E'\t', HEADER true);

-- Restore only the 4 affected rows (V1, V2, V3 are row-level; V4 touched 42)
UPDATE ssot_brochure.chapters c
   SET body_md = r.body_md
  FROM _chap_v10_rollback r
 WHERE c.slug = r.slug
   AND c.slug IN (
     'p3-13-phd-integration',
     'fm-07-olsen-tier-d',
     'london-handout'
   );

-- For V4 (TOC), restore all rows whose body_md ends in '## References {.unnumbered}'
UPDATE ssot_brochure.chapters c
   SET body_md = r.body_md
  FROM _chap_v10_rollback r
 WHERE c.slug = r.slug
   AND c.body_md LIKE '%## References {.unnumbered}%';

COMMIT;
SQL

# 2. Revert pipeline.rs
git checkout HEAD~1 -- src/pipeline.rs
cargo build --release

# 3. Rebuild
./target/release/trios-mcp-rag build-pdf --book-mode \
  --out-dir generated/out --build-dir generated/build \
```
