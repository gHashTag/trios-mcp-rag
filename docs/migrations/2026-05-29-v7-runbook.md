## Apply

```sql
BEGIN;

-- A5: strip trailing dot from bare DOIs
UPDATE ssot_brochure.chapters
SET body_md = regexp_replace(body_md,
  '(\m10\.[0-9]{4,5}/[A-Za-z0-9._/-]+?)\.($|\s|[)\],;:])',
  '\1\2', 'g');

-- A6: prefix bare arXiv NNNN.NNNNN with "arXiv:"
UPDATE ssot_brochure.chapters
SET body_md = regexp_replace(body_md,
  '\marXiv ([0-9]{4}\.[0-9]{4,6})\M', 'arXiv:\1', 'g');

-- A16a: Tier D + lowercase
UPDATE ssot_brochure.chapters
SET body_md = regexp_replace(body_md,
  '\mTier D(\s+[a-z—–-])', 'Tier-D\1', 'g');

-- A16b: Tier D + punctuation/whitespace
UPDATE ssot_brochure.chapters
SET body_md = regexp_replace(body_md,
  '\mTier D([\.,;:!?\s\)])', 'Tier-D\1', 'g');

-- A15: licence-and-attribution block in fm-01f
UPDATE ssot_brochure.chapters
SET body_md = replace(body_md,
  E'**Use of generative AI.**',
  E'**Licence and attribution.** ... \n\n**Use of generative AI.**')
WHERE slug = 'fm-01f-statements';

-- A14: insert fm-01g-errata-and-versions (handled via psycopg2 INSERT
-- ... ON CONFLICT (slug) DO UPDATE — see python snippet in audit doc)

-- Verification
SELECT 'Tier D remaining', count(*) FROM ssot_brochure.chapters WHERE body_md ~ '\mTier D\M';
SELECT 'DOI trailing dot remaining', count(*) FROM ssot_brochure.chapters
  WHERE body_md ~ '\m10\.[0-9]{4,5}/[A-Za-z0-9._/-]+?\.\s';

COMMIT;
```
